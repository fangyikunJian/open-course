在 `coursekit/_course.py` 里，`class Course` **之前**加模块级函数：

```python
import glob
import re


def merge_shards(data_dir, tag):
    """读一讲的内容：单文件 <tag>.json，或分片 <tag>.a.json/.b.json/…

    分片是为了让一讲的内容可以分几次写出来，而不必一次性持有整份文件 ——
    加厚后单讲预计 200 KB 以上，改一处就要整体重写，风险和成本都不对。
    （放映厅那边在 150 KB 单文件上撞过这面墙，解法相同。）

    片间页码重叠直接报错：静默覆盖会丢掉一整段讲解。
    """
    one = os.path.join(data_dir, tag + ".json")
    parts = sorted(glob.glob(os.path.join(data_dir, tag + ".[a-z].json")))
    if os.path.exists(one) and parts:
        raise SystemExit(
            "%s：单文件与分片同时存在（%s 与 %s）。留一种，否则会少读一半。"
            % (tag, os.path.basename(one),
               " ".join(os.path.basename(p) for p in parts)))
    files = [one] if os.path.exists(one) else parts
    if not files:
        return None

    notes, quiz, shards, owner = [], {}, [], {}
    for p in files:
        d = json.load(io.open(p, encoding="utf-8"))
        name = os.path.basename(p)
        shards.append(name)
        for n in d.get("notes") or []:
            r = n.get("r")
            if isinstance(r, list) and len(r) == 2:
                for pg in range(r[0], r[1] + 1):
                    if pg in owner:
                        raise SystemExit(
                            "%s：第 %d 页在 %s 与 %s 里重叠 —— "
                            "静默覆盖会丢掉一整段讲解。"
                            % (tag, pg, owner[pg], name))
                    owner[pg] = name
            notes.append(n)
        for k, v in (d.get("quiz") or {}).items():
            if k in quiz:
                raise SystemExit("%s：题库键 %s 在多个分片里重复" % (tag, k))
            quiz[k] = v
    notes.sort(key=lambda n: (n.get("r") or [10**9])[0])
    return {"notes": notes, "quiz": quiz, "_shards": shards}
```

```python
import re

# 「词」在本项目里只有这一个定义：剥掉 HTML 标签后，CJK 按单字、拉丁按词、
# 数字成串。密度闸门、覆盖率、报告里的每一个数都用它，不许另写一份 ——
# 两份实现会让同一个闸门在不同期给出不可比的数。
_TAG = re.compile(r"<[^>]+>")
_WORD = re.compile(r"[\u4e00-\u9fff]|[A-Za-z]+|[0-9.]+")


def wordcount(obj):
    """任意嵌套结构里的词数。"""
    if isinstance(obj, str):
        return len(_WORD.findall(_TAG.sub("", obj)))
    if isinstance(obj, dict):
        return sum(wordcount(v) for v in obj.values())
    if isinstance(obj, (list, tuple)):
        return sum(wordcount(v) for v in obj)
    return 0


def density(notes):
    """(词数, 有讲解的页数, 词/页)。分母是**被 notes 覆盖的页**，
    不是该讲总页数 —— 未覆盖页是增量构建页，本来就允许留空。"""
    pages = set()
    for n in notes:
        r = n.get("r")
        if isinstance(r, list) and len(r) == 2:
            pages.update(range(r[0], r[1] + 1))
    w = wordcount(notes)
    return w, len(pages), (w / len(pages) if pages else 0.0)
```

def field_cover(notes, key):
    """有该字段的页 / 有讲解的页。分母与密度一致。"""
    tot = cov = set()
    tot, cov = set(), set()
    for n in notes:
        r = n.get("r")
        if not (isinstance(r, list) and len(r) == 2):
            continue
        rng = range(r[0], r[1] + 1)
        tot.update(rng)
        if n.get(key):
            cov.update(rng)
    return (len(cov) / len(tot)) if tot else 0.0
```

def check_tbl(notes):
    """表格的结构问题。列数对不上表头，页面上会表现为「最后一格空着」，
    而不是报错 —— 所以必须机械查。"""
    out = []
    for n in notes:
        t = n.get("tbl")
        if not t:
            continue
        head = t.get("head")
        if not (isinstance(head, list) and head):
            out.append("p%s 的表格没有表头" % (n.get("r"),))
            continue
        for i, row in enumerate(t.get("rows") or []):
            if len(row) != len(head):
                out.append("p%s 表格第 %d 行 %d 格，表头 %d 列"
                           % (n.get("r"), i + 1, len(row), len(head)))
    return out
```
